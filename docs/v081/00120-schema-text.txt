/* Java 1.8 샘플 코드 */

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.net.HttpURLConnection;
import java.net.URI;
import java.net.URL;
import java.net.URLEncoder;

class ApiTest {
    public static void main(String[] args) throws Exception {

        String dataName = "데이터명";
        String serviceKey = "서비스키";
        String pageNo = "1";
        String numOfRows = "10";

        /* API를 호출하기 위한 URL 생성 */
        StringBuilder urlBuilder = new StringBuilder("https://www.safetydata.go.kr/V2/api/DSSP-IF-00120"); 
        /* API 키 */
        urlBuilder.append("?" + "serviceKey=" + serviceKey);
        /* 페이지 번호 */
        urlBuilder.append("&" + "pageNo=" + pageNo);
        /* 페이지당 데이터 수 */
        urlBuilder.append("&" + "numOfRows=" + numOfRows);


        System.out.println(urlBuilder.toString());
        URI uri = new URI(urlBuilder.toString());
        URL url = uri.toURL();

        /* API 호출하기 위한 HTTP 커넥션과 리더 생성 */
        HttpURLConnection connection =
                (HttpURLConnection) url.openConnection();
        connection.setRequestMethod("GET");

        BufferedReader reader;

        /* API 호출 */
        connection.connect();
        if (connection.getResponseCode() >= 200) {
            reader = new BufferedReader(new InputStreamReader(connection.getInputStream()));
        } else {
            System.err.println("Connection failed.");
            return;
        }

        /* API 응답에서 데이터 추출 */
        StringBuilder sb = new StringBuilder();
        String line;

        while ((line = reader.readLine()) != null) {
            sb.append(line);

            reader.close();
            connection.disconnect();
        }
    }
}

# C# 샘플 코드 샘플 코드 #

using System;
using System.Net.Http;
using System.Threading.Tasks;

namespace ConsoleApp1
{
    class Program
    {
        static HttpClient client = new HttpClient();

        static async Task Main(string[] args)
        {
	    string url = "https://www.safetydata.go.kr/V2/api/DSSP-IF-00120"; // URL
            url += "?serviceKey=" + "서비스키"; // service Key
            url += "&pageNo=1";
            url += "&numOfRows=10";

	     try
            {
                // 비동기 요청을 보냄
                HttpResponseMessage response = await client.GetAsync(url);
                
                // 응답이 성공적이면 데이터를 읽고 출력
                if (response.IsSuccessStatusCode)
                {
                    string results = await response.Content.ReadAsStringAsync();
                    Console.WriteLine(results);
                }
                else
                {
                    Console.WriteLine($"Error: {response.StatusCode}");
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Exception: {ex.Message}");
            }
        }
    }
}

# swift 코드 샘플 코드 #

import Foundation
	
class RequestAPI {
    static let shared = RequestAPI()
    private init() { }
    
    private let apiKey = Bundle.main.object(forInfoDictionaryKey: "서비스키") as? String
    
    func fetchData(){
        
        guard let apiKey = apiKey else { return }
        
        // API URL 설정
        guard let url = URL(string: "https://www.safetydata.go.kr/V2/api/DSSP-IF-00120?serviceKey=\(apiKey)&pageNo=1&numOfRows=10&") else{
            return
        }
        
        // URLSession 사용하여 데이터 요청
        let session = URLSession(configuration: .default)
        
        let task = session.dataTask(with: url) { data, response, error in
            if let error = error {
                return
            }
            
            guard let httpResponse = response as? HTTPURLResponse, httpResponse.statusCode == 200 else {
                print("데이터 가져오기 실패. 상태 코드: \((response as? HTTPURLResponse)?.statusCode ?? 0)")
                return
            }
            
            
            // 데이터 출력
            if let jsonString = String(data: data, encoding: .utf8) {
                print("응답 데이터: \(jsonString)")  // JSON 형식으로 출력
            }
        }
        
        task.resume()
    }
}
